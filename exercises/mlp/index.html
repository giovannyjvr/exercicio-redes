    """MLP escrito do zero em NumPy — forward, perda, backprop e update.

Nada aqui usa autodiff. Cada gradiente é derivado à mão e escrito explicitamente;
o NumPy entra apenas como álgebra linear.

**O núcleo é genérico e é reutilizado literalmente** nos Exercises 2, 3 e 4:

* ``layer_sizes`` é um argumento — uma lista como ``[2, 16, 1]`` ou
  ``[4, 32, 32, 3]``. Nada é fixado no código;
* :meth:`MLP.forward`, :meth:`MLP.backward` e :meth:`MLP.update` são **o mesmo
  código** para classificação binária e multiclasse. O que muda é um argumento:
  ``output="sigmoid"`` com entropia cruzada binária, ou ``output="softmax"`` com
  entropia cruzada categórica.

A razão de isso funcionar é a identidade que o Exercise 3 pede para derivar: tanto
para sigmoid + BCE quanto para softmax + entropia cruzada categórica,

.. math:: \\frac{\\partial L}{\\partial u^{(L)}} = \\frac{1}{N}(p - y)

onde :math:`p` é a saída do modelo e :math:`y` o alvo (one-hot no caso
multiclasse). O gradiente na saída tem a mesma forma nos dois casos, então o
backward não precisa saber qual perda está sendo usada.
"""

from __future__ import annotations

from typing import Any

import numpy as np

EPS = 1e-12  # evita log(0) nas perdas


def tanh(z: np.ndarray) -> np.ndarray:
    """Ativação das camadas escondidas."""
    return np.tanh(z)


def tanh_derivative(a: np.ndarray) -> np.ndarray:
    """Derivada da tanh escrita em função da **ativação**: :math:`1 - \\tanh^2(z)`."""
    return 1.0 - a**2


def sigmoid(u: np.ndarray) -> np.ndarray:
    """Sigmoide numericamente estável, para os dois sinais de ``u``."""
    positive = u >= 0
    out = np.empty_like(u, dtype=float)
    out[positive] = 1.0 / (1.0 + np.exp(-u[positive]))
    exp_u = np.exp(u[~positive])
    out[~positive] = exp_u / (1.0 + exp_u)
    return out


def softmax(u: np.ndarray) -> np.ndarray:
    """Softmax por linha, deslocado pelo máximo para não estourar o expoente."""
    shifted = u - u.max(axis=1, keepdims=True)
    exp_u = np.exp(shifted)
    return exp_u / exp_u.sum(axis=1, keepdims=True)


class MLP:
    """Perceptron multicamada com número arbitrário de camadas escondidas.

    Parameters
    ----------
    layer_sizes:
        Larguras, da entrada à saída. ``[2, 16, 1]`` é uma camada escondida de 16
        neurônios; ``[4, 32, 32, 3]`` são duas de 32 com 3 saídas.
    rng:
        Gerador usado na inicialização dos pesos.
    output:
        ``"sigmoid"`` (binário, com BCE) ou ``"softmax"`` (multiclasse, com
        entropia cruzada categórica).
    """

    def __init__(
        self,
        layer_sizes: list[int],
        rng: np.random.Generator,
        *,
        output: str = "sigmoid",
    ) -> None:
        if output not in {"sigmoid", "softmax"}:
            raise ValueError(f"saída desconhecida: {output!r}")
        self.layer_sizes = list(layer_sizes)
        self.output = output

        # Inicialização de Xavier/Glorot: escala 1/sqrt(fan_in) mantém a variância
        # das pré-ativações estável ao longo da profundidade. Com escala fixa, uma
        # rede mais funda saturaria a tanh já no forward inicial.
        self.weights: list[np.ndarray] = []
        self.biases: list[np.ndarray] = []
        for fan_in, fan_out in zip(self.layer_sizes[:-1], self.layer_sizes[1:]):
            self.weights.append(rng.normal(0.0, 1.0, size=(fan_in, fan_out)) / np.sqrt(fan_in))
            self.biases.append(np.zeros(fan_out))

    # ------------------------------------------------------------------ core
    # As três funções abaixo são o núcleo reutilizado sem alteração nos
    # Exercises 2, 3 e 4.

    def forward(self, X: np.ndarray) -> dict[str, Any]:
        """Passagem direta. Devolve as ativações de cada camada e a saída."""
        activations = [X]
        a = X
        # Camadas escondidas: pré-ativação afim seguida de tanh.
        for W, b in zip(self.weights[:-1], self.biases[:-1]):
            a = tanh(a @ W + b)
            activations.append(a)
        # Camada de saída: a pré-ativação é a mesma conta; só a não linearidade muda.
        u = a @ self.weights[-1] + self.biases[-1]
        probs = sigmoid(u) if self.output == "sigmoid" else softmax(u)
        return {"activations": activations, "u": u, "probs": probs}

    def backward(self, cache: dict[str, Any], Y: np.ndarray) -> tuple[list[np.ndarray], list[np.ndarray]]:
        """Retropropagação. Devolve ``(dW, db)``, um por camada.

        O gradiente na saída é ``(p - Y) / N`` nos dois casos — ver o docstring do
        módulo e a derivação no relatório. Daqui para trás o código é idêntico
        para binário e multiclasse.
        """
        activations, probs = cache["activations"], cache["probs"]
        n_samples = Y.shape[0]

        # Gradiente na pré-ativação da saída.
        delta = (probs - Y) / n_samples

        grads_w: list[np.ndarray] = [np.zeros_like(W) for W in self.weights]
        grads_b: list[np.ndarray] = [np.zeros_like(b) for b in self.biases]

        for layer in range(len(self.weights) - 1, -1, -1):
            a_prev = activations[layer]
            grads_w[layer] = a_prev.T @ delta
            grads_b[layer] = delta.sum(axis=0)
            if layer > 0:
                # Propaga para a camada anterior e atravessa a tanh.
                delta = (delta @ self.weights[layer].T) * tanh_derivative(activations[layer])

        return grads_w, grads_b

    def update(self, grads_w: list[np.ndarray], grads_b: list[np.ndarray], learning_rate: float) -> None:
        """Descida de gradiente simples, aplicada a todos os parâmetros."""
        for layer in range(len(self.weights)):
            self.weights[layer] -= learning_rate * grads_w[layer]
            self.biases[layer] -= learning_rate * grads_b[layer]

    # ------------------------------------------------------------------ perdas

    def loss(self, probs: np.ndarray, Y: np.ndarray) -> float:
        """Entropia cruzada binária ou categórica, média por amostra."""
        if self.output == "sigmoid":
            p = np.clip(probs, EPS, 1.0 - EPS)
            return float(-np.mean(Y * np.log(p) + (1.0 - Y) * np.log(1.0 - p)))
        p = np.clip(probs, EPS, 1.0)
        return float(-np.mean(np.sum(Y * np.log(p), axis=1)))

    # ------------------------------------------------------------------ uso

    def fit(
        self,
        X: np.ndarray,
        Y: np.ndarray,
        *,
        epochs: int,
        learning_rate: float,
    ) -> list[float]:
        """Treina em *full batch* e devolve a perda de treino de cada época."""
        history: list[float] = []
        for _ in range(epochs):
            cache = self.forward(X)
            history.append(self.loss(cache["probs"], Y))
            grads_w, grads_b = self.backward(cache, Y)
            self.update(grads_w, grads_b, learning_rate)
        return history

    def predict_proba(self, X: np.ndarray) -> np.ndarray:
        return self.forward(X)["probs"]

    def predict(self, X: np.ndarray) -> np.ndarray:
        """Rótulos preditos: limiar de 0.5 no binário, argmax no multiclasse."""
        probs = self.predict_proba(X)
        if self.output == "sigmoid":
            return (probs[:, 0] >= 0.5).astype(int)
        return probs.argmax(axis=1)

    def n_parameters(self) -> int:
        """Número total de pesos e vieses treináveis."""
        return int(sum(W.size for W in self.weights) + sum(b.size for b in self.biases))

    def architecture(self) -> str:
        """Descrição curta da arquitetura, para o relatório."""
        return "–".join(str(size) for size in self.layer_sizes)


def one_hot(y: np.ndarray, n_classes: int) -> np.ndarray:
    """Codifica rótulos inteiros como linhas one-hot."""
    encoded = np.zeros((y.shape[0], n_classes))
    encoded[np.arange(y.shape[0]), y] = 1.0
    return encoded

    """Exercise 1 — um passo de retropropagação, na mão.

Este módulo **não** usa autodiff nem biblioteca de rede neural: cada linha
corresponde a um passo da derivação feita no relatório, escrita explicitamente.
Ele existe para que os números citados no texto venham de uma execução real e
possam ser conferidos — a derivação em si está no `index.md`.

Rede: 2 entradas → 1 camada escondida de 2 neurônios → 1 saída, ``tanh`` nas duas
camadas, perda :math:`L = (y - \\hat{y})^2` sobre uma única amostra.
"""

from __future__ import annotations

from typing import Any

import numpy as np

# Valores dados pelo enunciado.
X = np.array([0.5, -0.2])
Y = 1.0
W1 = np.array([[0.3, -0.1], [0.2, 0.4]])
B1 = np.array([0.1, -0.2])
W2 = np.array([0.5, -0.3])
B2 = 0.2
ETA = 0.3


def run() -> dict[str, Any]:
    """Executa forward, backward e update, devolvendo todos os intermediários."""
    # --- A — Forward pass -------------------------------------------------
    z1 = W1 @ X + B1                     # pré-ativações da camada escondida
    h1 = np.tanh(z1)                     # ativações da camada escondida
    u2 = float(W2 @ h1 + B2)             # pré-ativação da saída
    y_hat = float(np.tanh(u2))           # saída
    loss = float((Y - y_hat) ** 2)       # L = (y - yhat)^2, com N = 1

    # --- B — Backward pass ------------------------------------------------
    # dL/dyhat = -2 (y - yhat)
    dL_dyhat = -2.0 * (Y - y_hat)
    # dyhat/du2 = 1 - tanh^2(u2) = 1 - yhat^2
    dL_du2 = dL_dyhat * (1.0 - y_hat**2)
    # Camada de saída: u2 = W2 · h1 + b2  =>  du2/dW2 = h1, du2/db2 = 1
    dL_dW2 = dL_du2 * h1
    dL_db2 = dL_du2
    # Propaga para trás: du2/dh1 = W2
    dL_dh1 = dL_du2 * W2
    # Atravessa a tanh da camada escondida: dh1/dz1 = 1 - h1^2
    dL_dz1 = dL_dh1 * (1.0 - h1**2)
    # Camada escondida: z1 = W1 x + b1  =>  dz1_i/dW1_ij = x_j, dz1_i/db1_i = 1
    dL_dW1 = np.outer(dL_dz1, X)
    dL_db1 = dL_dz1

    # --- C — Parameter update --------------------------------------------
    W2_new = W2 - ETA * dL_dW2
    b2_new = B2 - ETA * dL_db2
    W1_new = W1 - ETA * dL_dW1
    b1_new = B1 - ETA * dL_db1

    # Segundo forward, com os parâmetros atualizados, para confirmar a direção
    # da perda prevista na resposta do item C.
    z1_new = W1_new @ X + b1_new
    h1_new = np.tanh(z1_new)
    u2_new = float(W2_new @ h1_new + b2_new)
    y_hat_new = float(np.tanh(u2_new))
    loss_new = float((Y - y_hat_new) ** 2)

    return {
        "forward": {
            "z1": z1.tolist(), "h1": h1.tolist(),
            "u2": u2, "y_hat": y_hat, "loss": loss,
        },
        "backward": {
            "dL_dyhat": dL_dyhat, "dL_du2": dL_du2,
            "dL_dW2": dL_dW2.tolist(), "dL_db2": dL_db2,
            "dL_dh1": dL_dh1.tolist(), "dL_dz1": dL_dz1.tolist(),
            "dL_dW1": dL_dW1.tolist(), "dL_db1": dL_db1.tolist(),
        },
        "updated": {
            "W1": W1_new.tolist(), "b1": b1_new.tolist(),
            "W2": W2_new.tolist(), "b2": b2_new,
        },
        "second_forward": {
            "z1": z1_new.tolist(), "h1": h1_new.tolist(),
            "u2": u2_new, "y_hat": y_hat_new, "loss": loss_new,
            "loss_decreased": bool(loss_new < loss),
            "delta": loss_new - loss,
        },
    }


def report(result: dict[str, Any]) -> None:
    """Imprime todos os intermediários com 6 casas decimais."""
    f, b, u, s = result["forward"], result["backward"], result["updated"], result["second_forward"]
    print("A — Forward pass")
    print(f"  z1          = [{f['z1'][0]:.6f}, {f['z1'][1]:.6f}]")
    print(f"  h1          = [{f['h1'][0]:.6f}, {f['h1'][1]:.6f}]")
    print(f"  u2          = {f['u2']:.6f}")
    print(f"  y_hat       = {f['y_hat']:.6f}")
    print(f"  L           = {f['loss']:.6f}")
    print("\nB — Backward pass")
    print(f"  dL/dy_hat   = {b['dL_dyhat']:.6f}")
    print(f"  dL/du2      = {b['dL_du2']:.6f}")
    print(f"  dL/dW2      = [{b['dL_dW2'][0]:.6f}, {b['dL_dW2'][1]:.6f}]")
    print(f"  dL/db2      = {b['dL_db2']:.6f}")
    print(f"  dL/dh1      = [{b['dL_dh1'][0]:.6f}, {b['dL_dh1'][1]:.6f}]")
    print(f"  dL/dz1      = [{b['dL_dz1'][0]:.6f}, {b['dL_dz1'][1]:.6f}]")
    print(f"  dL/dW1      = [[{b['dL_dW1'][0][0]:.6f}, {b['dL_dW1'][0][1]:.6f}],"
          f" [{b['dL_dW1'][1][0]:.6f}, {b['dL_dW1'][1][1]:.6f}]]")
    print(f"  dL/db1      = [{b['dL_db1'][0]:.6f}, {b['dL_db1'][1]:.6f}]")
    print("\nC — Parameter update (eta = 0.3)")
    print(f"  W1          = [[{u['W1'][0][0]:.6f}, {u['W1'][0][1]:.6f}],"
          f" [{u['W1'][1][0]:.6f}, {u['W1'][1][1]:.6f}]]")
    print(f"  b1          = [{u['b1'][0]:.6f}, {u['b1'][1]:.6f}]")
    print(f"  W2          = [{u['W2'][0]:.6f}, {u['W2'][1]:.6f}]")
    print(f"  b2          = {u['b2']:.6f}")
    print(f"\n  segundo forward: y_hat = {s['y_hat']:.6f}, L = {s['loss']:.6f}"
          f" (variação {s['delta']:+.6f}; caiu: {s['loss_decreased']})")

def exercise_2(rng: np.random.Generator) -> dict[str, Any]:
    """Classificação binária que nenhuma reta resolve."""
    centers = np.array([[0.0, 0.0], [3.0, 3.0], [-3.0, -3.0]])
    X, c = make_blobs(
        n_samples=[500, 250, 250], centers=centers, cluster_std=1.2, random_state=SKLEARN_SEED
    )
    # Classe 0 = 1 cluster central; classe 1 = 2 clusters em lados opostos.
    y = (c > 0).astype(int)

    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, random_state=SKLEARN_SEED, stratify=y
    )

    # Figura 1 — dispersão dos 1000 pontos coloridos por y.
    fig, ax = plt.subplots(figsize=(7.5, 6.2))
    for label, color in zip((0, 1), CLASS_COLORS):
        pts = X[y == label]
        ax.scatter(pts[:, 0], pts[:, 1], s=12, alpha=0.6, color=color,
                   edgecolors="none", label=f"Classe {label}")
    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")
    ax.set_title("Figura 1 — Dois clusters da classe 1 cercando a classe 0\n"
                 "(make_blobs, 1000 pontos, cluster_std = 1.2)")
    ax.legend(loc="upper left")
    ax.grid(alpha=0.2, linestyle=":")
    fig1 = save_figure(fig, EXPECTED_FIGURES[0])

    # B — baseline linear. É a referência, não o modelo.
    baseline = LogisticRegression(max_iter=1000).fit(X_train, y_train)
    baseline_accuracy = accuracy(y_test, baseline.predict(X_test))

    # Geometria do baseline, para o item D. Duas perguntas que o texto responde:
    # (i) onde a fronteira logística se posiciona, e (ii) qual seria a melhor
    # acurácia alcançável por *alguma* reta — obtida por busca exaustiva em
    # ângulo e deslocamento, não por otimização de alguma perda substituta.
    w_base, b_base = baseline.coef_[0], float(baseline.intercept_[0])
    baseline_geometry = {
        "w": w_base.tolist(),
        "b": b_base,
        "distance_to_origin": float(abs(b_base) / np.linalg.norm(w_base)),
        "recall_class_0": float(np.mean(baseline.predict(X[y == 0]) == 0)),
        "recall_class_1": float(np.mean(baseline.predict(X[y == 1]) == 1)),
    }
    best_line = 0.0
    for angle in np.linspace(0.0, np.pi, 721):
        direction = np.array([np.cos(angle), np.sin(angle)])
        projection = X_test @ direction
        for offset in np.unique(np.round(projection, 3)):
            for sign in (1.0, -1.0):
                acc = accuracy(y_test, (sign * (projection - offset) >= 0).astype(int))
                best_line = max(best_line, acc)
    baseline_geometry["best_possible_line"] = float(best_line)

    # C — MLP do zero.
    layer_sizes = [2, 16, 1]
    learning_rate, epochs = 0.5, 3000
    model = MLP(layer_sizes, rng, output="sigmoid")
    history = model.fit(
        X_train, y_train.reshape(-1, 1).astype(float), epochs=epochs, learning_rate=learning_rate
    )
    test_accuracy = accuracy(y_test, model.predict(X_test))
    train_accuracy = accuracy(y_train, model.predict(X_train))

    # Figura 2 — perda de treino por época.
    fig, ax = plt.subplots(figsize=(8.0, 5.0))
    ax.plot(range(1, epochs + 1), history, color="#1f77b4", linewidth=1.8,
            label="Entropia cruzada binária (treino)")
    ax.set_xlabel("Época")
    ax.set_ylabel("Perda média por amostra (nats)")
    ax.set_title(f"Figura 2 — Perda de treino, MLP {model.architecture()}\n"
                 f"(lr = {learning_rate}, {epochs} épocas, full batch)")
    ax.grid(alpha=0.25, linestyle=":")
    ax.legend()
    fig2 = save_figure(fig, EXPECTED_FIGURES[1])

    # Figura 3 — fronteira de decisão avaliada numa malha, sobre os pontos de teste.
    pad = 1.0
    gx = np.linspace(X[:, 0].min() - pad, X[:, 0].max() + pad, 400)
    gy = np.linspace(X[:, 1].min() - pad, X[:, 1].max() + pad, 400)
    mesh_x, mesh_y = np.meshgrid(gx, gy)
    grid = np.column_stack([mesh_x.ravel(), mesh_y.ravel()])
    zones = model.predict(grid).reshape(mesh_x.shape)

    fig, ax = plt.subplots(figsize=(8.0, 6.5))
    ax.contourf(mesh_x, mesh_y, zones, levels=[-0.5, 0.5, 1.5],
                colors=CLASS_COLORS[:2], alpha=0.18)
    ax.contour(mesh_x, mesh_y, zones, levels=[0.5], colors="black", linewidths=1.6)
    for label, color in zip((0, 1), CLASS_COLORS):
        pts = X_test[y_test == label]
        ax.scatter(pts[:, 0], pts[:, 1], s=20, alpha=0.85, color=color,
                   edgecolors="white", linewidths=0.4, label=f"Teste — classe {label}")
    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")
    ax.set_title(f"Figura 3 — Regiões de decisão do MLP sobre os pontos de teste\n"
                 f"(acurácia de teste {test_accuracy * 100:.2f}%)")
    ax.legend(loc="upper left")
    fig3 = save_figure(fig, EXPECTED_FIGURES[2])

    print("\nExercise 2 — classificação binária")
    print(f"  baseline LogisticRegression: acurácia de teste = {baseline_accuracy * 100:.2f}%")
    print(f"    fronteira a {baseline_geometry['distance_to_origin']:.4f} da origem |"
          f" acerta {baseline_geometry['recall_class_0'] * 100:.2f}% da classe 0 e"
          f" {baseline_geometry['recall_class_1'] * 100:.2f}% da classe 1")
    print(f"    melhor reta possível (busca exaustiva no teste) ="
          f" {baseline_geometry['best_possible_line'] * 100:.2f}%")
    print(f"  MLP {model.architecture()} | lr = {learning_rate} | épocas = {epochs}"
          f" | parâmetros = {model.n_parameters()}")
    print(f"  acurácia: treino = {train_accuracy * 100:.2f}%, teste = {test_accuracy * 100:.2f}%")
    print(f"  perda de treino: inicial = {history[0]:.6f}, final = {history[-1]:.6f}")

    return {
        "n_samples": int(len(X)),
        "n_train": int(len(X_train)),
        "n_test": int(len(X_test)),
        "baseline_accuracy": baseline_accuracy,
        "baseline_geometry": baseline_geometry,
        "architecture": model.architecture(),
        "layer_sizes": layer_sizes,
        "learning_rate": learning_rate,
        "epochs": epochs,
        "n_parameters": model.n_parameters(),
        "train_accuracy": train_accuracy,
        "test_accuracy": test_accuracy,
        "loss_first": history[0],
        "loss_final": history[-1],
        "figures": [fig1, fig2, fig3],
    }

def make_multiclass() -> tuple[np.ndarray, ...]:
    """Dataset dos Exercises 3 e 4 — gerado uma vez e reutilizado."""
    X, y = make_classification(
        n_samples=1500, n_features=4, n_informative=4, n_redundant=0, n_repeated=0,
        n_classes=3, n_clusters_per_class=2, class_sep=1.0, random_state=SKLEARN_SEED,
    )
    return train_test_split(X, y, test_size=0.2, random_state=SKLEARN_SEED, stratify=y)


def train_multiclass(
    layer_sizes: list[int], rng: np.random.Generator, data: tuple[np.ndarray, ...],
    *, learning_rate: float, epochs: int,
) -> dict[str, Any]:
    """Treina a rede multiclasse — o mesmo núcleo do Exercise 2, com softmax."""
    X_train, X_test, y_train, y_test = data
    model = MLP(layer_sizes, rng, output="softmax")
    history = model.fit(
        X_train, one_hot(y_train, 3), epochs=epochs, learning_rate=learning_rate
    )
    return {
        "model": model,
        "history": history,
        "train_accuracy": accuracy(y_train, model.predict(X_train)),
        "test_accuracy": accuracy(y_test, model.predict(X_test)),
        "loss_final": history[-1],
    }


def exercise_3(rng: np.random.Generator, data: tuple[np.ndarray, ...]) -> dict[str, Any]:
    """Multiclasse com o mesmo núcleo, trocando só saída e perda."""
    X_train, X_test, y_train, y_test = data
    layer_sizes, learning_rate, epochs = [4, 32, 3], 0.5, 4000

    run = train_multiclass(layer_sizes, rng, data, learning_rate=learning_rate, epochs=epochs)
    model, history = run["model"], run["history"]

    # O scikit-learn >= 1.9 removeu o parâmetro multi_class; o padrão para 3
    # classes já é a regressão logística multinomial (softmax).
    baseline = LogisticRegression(max_iter=1000).fit(X_train, y_train)
    baseline_accuracy = accuracy(y_test, baseline.predict(X_test))

    # Figura 4 — perda de treino por época.
    fig, ax = plt.subplots(figsize=(8.0, 5.0))
    ax.plot(range(1, epochs + 1), history, color="#2ca02c", linewidth=1.8,
            label="Entropia cruzada categórica (treino)")
    ax.set_xlabel("Época")
    ax.set_ylabel("Perda média por amostra (nats)")
    ax.set_title(f"Figura 4 — Perda de treino, MLP {model.architecture()}\n"
                 f"(lr = {learning_rate}, {epochs} épocas, 3 classes)")
    ax.grid(alpha=0.25, linestyle=":")
    ax.legend()
    fig4 = save_figure(fig, EXPECTED_FIGURES[3])

    # Figura 5 — matriz de confusão no teste.
    cm = confusion_matrix(y_test, model.predict(X_test))
    fig, ax = plt.subplots(figsize=(6.8, 5.8))
    im = ax.imshow(cm, cmap="Blues")
    for i in range(cm.shape[0]):
        for j in range(cm.shape[1]):
            ax.text(j, i, str(cm[i, j]), ha="center", va="center",
                    color="white" if cm[i, j] > cm.max() / 2 else "black", fontsize=12)
    ax.set_xticks(range(3), [f"Classe {k}" for k in range(3)])
    ax.set_yticks(range(3), [f"Classe {k}" for k in range(3)])
    ax.set_xlabel("Classe prevista")
    ax.set_ylabel("Classe verdadeira")
    ax.set_title(f"Figura 5 — Matriz de confusão no teste\n"
                 f"(acurácia {run['test_accuracy'] * 100:.2f}%, {len(y_test)} amostras)")
    fig.colorbar(im, ax=ax, label="Nº de amostras")
    fig5 = save_figure(fig, EXPECTED_FIGURES[4])

    # Par de classes mais confundido: soma os dois sentidos fora da diagonal.
    pairs = []
    for i in range(3):
        for j in range(i + 1, 3):
            pairs.append({"i": i, "j": j, "errors": int(cm[i, j] + cm[j, i]),
                          "i_to_j": int(cm[i, j]), "j_to_i": int(cm[j, i])})
    worst = max(pairs, key=lambda p: p["errors"])
    total_errors = int(cm.sum() - np.trace(cm))

    print("\nExercise 3 — multiclasse")
    print(f"  baseline LogisticRegression: acurácia de teste = {baseline_accuracy * 100:.2f}%")
    print(f"  MLP {model.architecture()} | lr = {learning_rate} | épocas = {epochs}"
          f" | parâmetros = {model.n_parameters()}")
    print(f"  acurácia: treino = {run['train_accuracy'] * 100:.2f}%,"
          f" teste = {run['test_accuracy'] * 100:.2f}%")
    print(f"  matriz de confusão:\n{cm}")
    print(f"  par mais confundido: classes {worst['i']} e {worst['j']}"
          f" ({worst['errors']} erros de {total_errors}:"
          f" {worst['i']}->{worst['j']} = {worst['i_to_j']},"
          f" {worst['j']}->{worst['i']} = {worst['j_to_i']})")

    return {
        "n_train": int(len(X_train)),
        "n_test": int(len(X_test)),
        "baseline_accuracy": baseline_accuracy,
        "architecture": model.architecture(),
        "layer_sizes": layer_sizes,
        "learning_rate": learning_rate,
        "epochs": epochs,
        "n_parameters": model.n_parameters(),
        "train_accuracy": run["train_accuracy"],
        "test_accuracy": run["test_accuracy"],
        "loss_first": history[0],
        "loss_final": history[-1],
        "confusion_matrix": cm.tolist(),
        "class_pairs": pairs,
        "worst_pair": worst,
        "total_errors": total_errors,
        "history": history,
        "figures": [fig4, fig5],
    }

def exercise_4(data: tuple[np.ndarray, ...], shallow: dict[str, Any]) -> dict[str, Any]:
    """Comparação controlada: só a profundidade muda."""
    learning_rate = shallow["learning_rate"]
    epochs = shallow["epochs"]
    shallow_sizes = shallow["layer_sizes"]
    deep_sizes = [4, 32, 32, 3]

    # Mesmos dados, mesmo orçamento, mesma taxa; repetido com 3 sementes.
    runs: dict[str, list[dict[str, Any]]] = {"shallow": [], "deep": []}
    curves: dict[str, list[float]] = {}
    for seed in DEPTH_SEEDS:
        for name, sizes in (("shallow", shallow_sizes), ("deep", deep_sizes)):
            run = train_multiclass(
                sizes, np.random.default_rng(seed), data,
                learning_rate=learning_rate, epochs=epochs,
            )
            runs[name].append({
                "seed": seed,
                "test_accuracy": run["test_accuracy"],
                "train_accuracy": run["train_accuracy"],
                "loss_final": run["loss_final"],
                "n_parameters": run["model"].n_parameters(),
                "architecture": run["model"].architecture(),
            })
            if seed == DEPTH_SEEDS[0]:
                curves[name] = run["history"]

    def summarize(name: str) -> dict[str, Any]:
        rows = runs[name]
        return {
            "architecture": rows[0]["architecture"],
            "hidden_layers": len(rows[0]["architecture"].split("–")) - 2,
            "n_parameters": rows[0]["n_parameters"],
            "test_accuracy_mean": float(np.mean([r["test_accuracy"] for r in rows])),
            "test_accuracy_std": float(np.std([r["test_accuracy"] for r in rows], ddof=1)),
            "train_accuracy_mean": float(np.mean([r["train_accuracy"] for r in rows])),
            "loss_final_mean": float(np.mean([r["loss_final"] for r in rows])),
            "per_seed": rows,
        }

    summary = {"shallow": summarize("shallow"), "deep": summarize("deep")}
    difference = summary["deep"]["test_accuracy_mean"] - summary["shallow"]["test_accuracy_mean"]

    # Figura 6 — curvas de perda das duas redes nos mesmos eixos.
    fig, ax = plt.subplots(figsize=(8.5, 5.2))
    ax.plot(range(1, epochs + 1), curves["shallow"], color="#2ca02c", linewidth=1.8,
            label=f"Exercise 3 — {summary['shallow']['architecture']} (1 camada escondida)")
    ax.plot(range(1, epochs + 1), curves["deep"], color="#9467bd", linewidth=1.8,
            label=f"Exercise 4 — {summary['deep']['architecture']} (2 camadas escondidas)")
    ax.set_xlabel("Época")
    ax.set_ylabel("Perda média por amostra (nats)")
    ax.set_title("Figura 6 — Perda de treino: uma camada escondida contra duas\n"
                 f"(mesmos dados, mesma lr = {learning_rate}, mesmas {epochs} épocas, semente {DEPTH_SEEDS[0]})")
    ax.grid(alpha=0.25, linestyle=":")
    ax.legend()
    fig6 = save_figure(fig, EXPECTED_FIGURES[5])

    print("\nExercise 4 — a profundidade ajuda?")
    for name in ("shallow", "deep"):
        s = summary[name]
        accs = ", ".join(f"{r['test_accuracy'] * 100:.2f}%" for r in s["per_seed"])
        print(f"  {s['architecture']:<12} | camadas escondidas = {s['hidden_layers']}"
              f" | parâmetros = {s['n_parameters']:>5}"
              f" | perda final média = {s['loss_final_mean']:.6f}"
              f" | teste médio = {s['test_accuracy_mean'] * 100:.2f}%"
              f" (±{s['test_accuracy_std'] * 100:.2f}) [{accs}]")
    print(f"  diferença (profunda − rasa) = {difference * 100:+.2f} pontos percentuais")

    return {
        "seeds": list(DEPTH_SEEDS),
        "learning_rate": learning_rate,
        "epochs": epochs,
        "summary": summary,
        "difference_mean": difference,
        "figures": [fig6],
    }

    Exercise 1

    Rede com 2 entradas, 1 camada escondida de 2 neurônios e 1 saída, com \(\tanh\) nas duas camadas e perda \(L = (y - \hat{y})^2\) sobre uma única amostra (\(N = 1\)).

    Parâmetro Valor
    Entrada e alvo \(\mathbf{x} = [0.5,\ -0.2]\), \(y = 1.0\)
    \(\mathbf{W}^{(1)}\) \(\begin{bmatrix} 0.3 & -0.1 \\ 0.2 & 0.4 \end{bmatrix}\)
    \(\mathbf{b}^{(1)}\) \([0.1,\ -0.2]\)
    \(\mathbf{W}^{(2)}\) \([0.5,\ -0.3]\)
    \(b^{(2)}\) \(0.2\)
    \(\eta\) \(0.3\)

    Derivada usada o tempo todo: \(\dfrac{d}{du}\tanh(u) = 1 - \tanh^2(u)\).

    A — Forward pass

    Pré-ativações da camada escondida, \(\mathbf{z}^{(1)} = \mathbf{W}^{(1)}\mathbf{x} + \mathbf{b}^{(1)}\):

    \[ z^{(1)}_1 = (0.3)(0.5) + (-0.1)(-0.2) + 0.1 = 0.15 + 0.02 + 0.1 = \mathbf{0.270000} \]
    \[ z^{(1)}_2 = (0.2)(0.5) + (0.4)(-0.2) + (-0.2) = 0.10 - 0.08 - 0.20 = \mathbf{-0.180000} \]

    B — Backward pass

    Começo da cadeia. Com \(L = (y - \hat{y})^2\):

    \[ \frac{\partial L}{\partial \hat{y}} = -2\,(y - \hat{y}) = -2\,(1 - 0.367247) = \mathbf{-1.265507} \]

    Atravessando a \(\tanh\) da saída. Como \(\hat{y} = \tanh(u^{(2)})\), vale \(\partial\hat{y}/\partial u^{(2)} = 1 - \hat{y}^2 = 1 - 0.134870 = 0.865130\):

    \[ \frac{\partial L}{\partial u^{(2)}} = \frac{\partial L}{\partial \hat{y}}\,(1 - \hat{y}^2) = (-1.265507)(0.865130) = \mathbf{-1.094828} \]

    Camada de saída. De \(u^{(2)} = \sum_i W^{(2)}_i h^{(1)}_i + b^{(2)}\) saem \(\partial u^{(2)}/\partial W^{(2)}_i = h^{(1)}_i\) e \(\partial u^{(2)}/\partial b^{(2)} = 1\):

    \[ \frac{\partial L}{\partial \mathbf{W}^{(2)}} = \frac{\partial L}{\partial u^{(2)}}\,\mathbf{h}^{(1)} = (-1.094828)\,[0.263625,\ -0.178081] = \mathbf{[-0.288624,\ 0.194968]} \]
    \[ \frac{\partial L}{\partial b^{(2)}} = \frac{\partial L}{\partial u^{(2)}} = \mathbf{-1.094828} \]

    Propagando para trás. De \(\partial u^{(2)}/\partial h^{(1)}_i = W^{(2)}_i\):

    \[ \frac{\partial L}{\partial \mathbf{h}^{(1)}} = \frac{\partial L}{\partial u^{(2)}}\,\mathbf{W}^{(2)} = (-1.094828)\,[0.5,\ -0.3] = \mathbf{[-0.547414,\ 0.328448]} \]

    Camada escondida. De \(z^{(1)}_i = \sum_j W^{(1)}_{ij} x_j + b^{(1)}_i\) saem \(\partial z^{(1)}_i/\partial W^{(1)}_{ij} = x_j\) e \(\partial z^{(1)}_i/\partial b^{(1)}_i = 1\), ou seja, o gradiente dos pesos é o produto externo \(\frac{\partial L}{\partial \mathbf{z}^{(1)}} \mathbf{x}^\top\):

    D — Analysis

    Por que o baseline linear fica abaixo de 50 % enquanto o MLP passa de 90 %.

    Olhe a Figura 3. A região que a rede atribui à classe 0 é uma faixa (uma tira diagonal) limitada por duas fronteiras aproximadamente paralelas, e a classe 1 ocupa os dois semiplanos que sobram, um de cada lado da faixa. Essa é a forma que a geometria dos dados exige: a classe 1 tem dois clusters, em \([3,3]\) e \([-3,-3]\), e eles estão em lados opostos do cluster da classe 0, centrado na origem.

    Uma única reta não consegue produzir essa forma, e a razão é exata: uma reta divide o plano em dois semiplanos, cada um convexo e conexo. Mas a região da classe 1 que os dados pedem é a união de dois conjuntos disjuntos, separados pela faixa da classe 0 no meio. Nenhum semiplano é a união de dois pedaços desconexos. É preciso de pelo menos duas fronteiras lineares — uma de cada lado da faixa — e é exatamente isso que os 16 neurônios escondidos fornecem: cada neurônio contribui com uma fronteira linear, e a camada de saída as combina.

    Por que abaixo de 50 %, e não exatamente 50 %. Este é o detalhe que torna o resultado interessante, e ele exige separar duas coisas: o que uma reta poderia fazer e o que a regressão logística faz.

    Uma busca exaustiva sobre todas as retas — varrendo 721 ângulos e todos os deslocamentos possíveis — mostra que a melhor reta alcança 75.50 % no conjunto de teste. Faz sentido: ela isola um dos clusters da classe 1 e chama todo o resto de classe 0, acertando 250 + 500 de 1000 pontos. Então o teto do linear é 75 %, não 50 %.

    A regressão logística, porém, não otimiza acurácia — ela minimiza a perda logística. E os dois clusters da classe 1 estão em posições simétricas em torno da origem (\([3,3]\) e \([-3,-3]\)): cada um puxa a fronteira para um lado, e os puxões se cancelam. O resultado medido é uma fronteira praticamente passando pela origem: a distância dela até a origem é de apenas 0.1044, ou seja, ela corta o cluster da classe 0 bem no meio. A consequência aparece nos acertos por classe:

    Classe Acertos do baseline linear
    0 (cluster central) 53.40 %
    1 (dois clusters) 50.00 %

    Ela divide a classe 0 quase ao meio e a classe 1 exatamente ao meio — um cluster de cada lado. Daí os 47.50 %, abaixo do acaso. A lição é que simetria não deixa o classificador linear apenas ruim: deixa-o pior que uma moeda, porque a perda que ele minimiza o leva ativamente para o pior lugar possível da família de retas, distante dos 75.50 % que a melhor reta atingiria.

    B — Adapt the network

    O mesmo núcleo do Exercise 2, com três mudanças que são argumentos, não estrutura: a saída passa a ter 3 unidades, a não linearidade final passa a ser softmax, e a perda passa a ser entropia cruzada categórica.

    \[ p_k = \frac{e^{u_k}}{\sum_{j=1}^{3} e^{u_j}}, \qquad L = -\sum_{k=1}^{3} y_k \log p_k \]

    Derivação de \(\partial L / \partial u\)

    Esta é a conta que permite que o backward seja literalmente o mesmo código nos dois exercícios. Primeiro, a derivada do softmax:

    \[ \frac{\partial p_k}{\partial u_i} = p_k\,(\delta_{ki} - p_i), \]

    onde \(\delta_{ki}\) vale 1 se \(k = i\) e 0 caso contrário. Agora a regra da cadeia, somando sobre todas as saídas (pois \(u_i\) afeta todas as \(p_k\) pelo denominador compartilhado):

    \[ \frac{\partial L}{\partial u_i} = \sum_{k} \frac{\partial L}{\partial p_k}\frac{\partial p_k}{\partial u_i} = \sum_{k} \left(-\frac{y_k}{p_k}\right) p_k\,(\delta_{ki} - p_i) = -\sum_{k} y_k\,(\delta_{ki} - p_i) \]

    Abrindo a soma, o termo com \(\delta_{ki}\) sobrevive só em \(k = i\):

    \[ \frac{\partial L}{\partial u_i} = -y_i + p_i \sum_{k} y_k = -y_i + p_i \cdot 1 = \boxed{\,p_i - y_i\,} \]

    usando que \(\sum_k y_k = 1\) num vetor one-hot. O \(p_k\) do denominador cancela com o \(p_k\) da derivada do softmax — é daí que vem a simplicidade famosa do resultado.

    E o caso binário dá a mesma forma. Com \(p = \sigma(u)\) e \(L = -[y\log p + (1-y)\log(1-p)]\):

    \[ \frac{\partial L}{\partial p} = \frac{p - y}{p\,(1-p)}, \qquad \frac{\partial p}{\partial u} = p\,(1-p) \;\;\Longrightarrow\;\; \frac{\partial L}{\partial u} = p - y \]

    Por isso o backward() de mlp.py não precisa saber qual perda está em uso: ele começa de delta = (probs - Y) / N e segue idêntico. A reutilização verbatim que o item extra pede não é um truque de organização — ela é consequência direta desta identidade.

    1. Cada classe é desconexa. Como cada classe é a união de dois clusters que podem cair em cantos distantes do hipercubo, a região de decisão de uma classe não é um bloco único. A rede precisa aprender duas ilhas por classe — é o mesmo problema do Exercise 2, agora em 4D e com três rótulos.

A fronteira necessária é uma superfície que envolve algumas regiões convexas — complexa o bastante para derrotar um modelo linear (os 66.33 % do baseline provam isso), mas rasa em termos de composição. Pelo teorema da aproximação universal, uma camada escondida com largura suficiente já representa qualquer fronteira dessas; e 32 neurônios em 4D são capazes de carregar bem mais do que 6 blobs exigem. A segunda camada não tem o que acrescentar em representação — a primeira já é expressiva o suficiente.

    O que a segunda camada custa. Dois preços, ambos visíveis nos números:

    1. Estatístico. 1315 parâmetros contra 259, para 1200 amostras de treino. A rede profunda tem mais parâmetros do que exemplos. Sem regularização, a folga vira ajuste a ruído: os 16.55 pontos de distância entre treino e teste.
    2. De otimização. Gradientes atravessam mais uma \(\tanh\) antes de chegar à primeira camada, e cada passagem multiplica por \(1 - \tanh^2 \le 1\) — o sinal que chega às camadas iniciais encolhe. Com o mesmo orçamento e a mesma taxa, a rede profunda não está resolvendo o mesmo problema de otimização com mais ferramentas; está resolvendo um problema pior condicionado.

    Que dado premiaria a profundidade. Dados com estrutura composicional e hierárquica, em que features úteis se constroem a partir de features úteis mais simples. Imagens são o exemplo canônico: pixels → bordas → texturas → partes → objetos, e cada nível reaproveita o anterior. Uma rede rasa teria de representar "objeto" diretamente a partir de pixels, o que exige largura exponencialmente maior que a profundidade equivalente. O mesmo vale para linguagem (caracteres → palavras → sintagmas → sentido) e para funções que são composições de muitos passos. Aqui, ao contrário, as 4 features já são as variáveis informativas — não existe hierarquia a explorar, e a profundidade só adiciona parâmetros.

    Vale a precisão final: o resultado é que a profundidade não ajuda neste conjunto, com este orçamento e sem regularização. Não é uma afirmação sobre profundidade em geral.
